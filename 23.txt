Q1 [10 Marks] Create weather data and a correlation Heat Map.
Solution — Q1
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
df = pd.DataFrame({
    "Study Hours":[2,4,5,7,8,10],
    "Attendance":[60,70,75,85,90,95],
    "Assignment Score":[55,65,70,80,88,95],
    "Exam Score":[50,62,68,78,85,92]
})
corr = df.corr()
print(corr)
sns.heatmap(corr,annot=True,cmap="coolwarm")
plt.title("Correlation Heat Map")
plt.show()
Q2 [20 Marks] Apply K-Means on Student data using Attendance and Marks with K = 3.
Solution — Q2 (Option A)
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
df=pd.read_csv("student_dataset_v2.csv").dropna()
X=df[["Attendance","Marks"]]
Xsc=StandardScaler().fit_transform(X)
model=KMeans(n_clusters=3,n_init=10,random_state=42)
df["Cluster"]=model.fit_predict(Xsc)
centers=model.cluster_centers_
plt.scatter(Xsc[:,0],Xsc[:,1],c=df["Cluster"],cmap="viridis")
plt.scatter(centers[:,0],centers[:,1],c="red",marker="X",s=150)
plt.xlabel("Attendance"); plt.ylabel("Marks")
plt.title("K-Means Clustering (K=3)")
plt.show()
print(df[["Attendance","Marks","Cluster"]])

Q2 [20 Marks] Perform Feature Selection and PCA on Loan_application.csv.
Solution — Q2 (Option B)
import pandas as pd
from sklearn.feature_selection import SelectKBest,chi2
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
df=pd.DataFrame({
 "Application_ID":range(1,11),
 "Age":[22,25,29,31,35,40,42,45,50,55],
 "Income":[25000,30000,42000,50000,55000,65000,70000,80000,90000,100000],
 "Credit_Score":[620,650,680,700,720,740,760,780,800,820],
 "Loan_Amount":[10000,15000,20000,25000,30000,35000,40000,45000,50000,60000],
 "Employment_Years":[1,2,3,4,5,7,8,10,12,15],
 "Debt_Ratio":[.5,.45,.4,.38,.35,.3,.28,.25,.22,.2],
 "Loan_Status":["Rejected","Rejected","Approved","Approved","Approved",
                "Approved","Approved","Approved","Approved","Approved"]
})
X=df.drop(columns=["Application_ID","Loan_Status"])
y=(df["Loan_Status"]=="Approved").astype(int)
selected=SelectKBest(chi2,k=3).fit_transform(X,y)
print("Selected features:\n",selected)
scaled=StandardScaler().fit_transform(X)
pca=PCA(n_components=2)
reduced=pca.fit_transform(scaled)
print("\nPCA result:\n",reduced)
print("Explained variance:",pca.explained_variance_ratio_)
