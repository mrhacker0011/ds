Q2 [20 Marks] Create city data, a population Treemap and a 3D Scatter Plot.
Solution — Q2 (Option A)
import pandas as pd
import matplotlib.pyplot as plt
import squarify
df=pd.DataFrame({
 "City":["Pune","Mumbai","Nashik","Nagpur","Kolhapur"],
 "Area":[331,603,259,228,66],
 "Temperature":[25,28,24,27,23],
 "Rainfall":[722,2400,700,1100,900],
 "Population_Density":[5200,20000,3200,4500,6000]
})
squarify.plot(sizes=df["Population_Density"],label=df["City"],alpha=.8)
plt.title("Population Density Treemap")
plt.axis("off"); plt.show()
fig=plt.figure(figsize=(8,5))
ax=fig.add_subplot(111,projection="3d")
ax.scatter(df["Area"],df["Temperature"],df["Rainfall"],s=70)
ax.set_xlabel("Area"); ax.set_ylabel("Temperature"); ax.set_zlabel("Rainfall")
plt.title("City 3D Scatter Plot")
plt.show()

Q2 [20 Marks] Perform Grocery Apriori for support 0.3 and 0.4.
Solution — Q2 (Option B)
import pandas as pd
from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import apriori,association_rules
transactions=[
["Basmati Rice","Toor Dal","Turmeric Powder","Cooking Oil"],
["Atta","Sugar","Tea","Milk"],
["Rice","Moong Dal","Salt","Cooking Oil"],
["Wheat Flour","Chickpeas","Turmeric Powder","Cumin Seeds"],
["Poha","Peanuts","Onions","Cooking Oil"],
["Basmati Rice","Paneer","Garam Masala","Tomatoes"],
["Atta","Potatoes","Onions","Cooking Oil"]]
te=TransactionEncoder()
df=pd.DataFrame(te.fit(transactions).transform(transactions),
                columns=te.columns_)
for support in [.3,.4]:
    freq=apriori(df,min_support=support,use_colnames=True)
    rules=association_rules(freq,metric="confidence",min_threshold=.5)
    print("\nSupport:",support)
    print(freq)
    print(rules[["antecedents","consequents","support","confidence"]])
