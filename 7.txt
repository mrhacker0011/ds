Q2 [20 Marks] Implement KNN on the Student dataset and classify performance categories.
Solution — Q2 (Option A)
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score,precision_score,recall_score,f1_score,confusion_matrix
df=pd.read_csv("student_dataset_v2.csv").dropna()
target=df.columns[-1]
X=pd.get_dummies(df.drop(columns=[target]),drop_first=True)
y=df[target].astype("category").cat.codes
X=StandardScaler().fit_transform(X)
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,
                                  random_state=42,stratify=y)
model=KNeighborsClassifier(n_neighbors=5).fit(Xtr,ytr)
pred=model.predict(Xte)
print("Accuracy:",accuracy_score(yte,pred))
print("Precision:",precision_score(yte,pred,average="weighted"))
print("Recall:",recall_score(yte,pred,average="weighted"))
print("F1:",f1_score(yte,pred,average="weighted"))
print("Confusion Matrix:\n",confusion_matrix(yte,pred))

Q2 [20 Marks] Perform Grocery market basket analysis using Apriori for support 0.3 and 0.4.
Solution — Q2 (Option B)
import pandas as pd
from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import apriori,association_rules
transactions=[
["Basmati Rice","Toor Dal","Turmeric Powder","Cooking Oil"],
["Atta","Sugar","Tea","Milk"],
["Rice","Moong Dal","Salt","Cooking Oil"],
["Wheat Flour","Chickpeas","Turmeric Powder","Cumin Seeds"],
["Poha","Peanuts","Onions","Cooking Oil"],
["Basmati Rice","Paneer","Garam Masala","Tomatoes"],
["Atta","Potatoes","Onions","Cooking Oil"]]
te=TransactionEncoder()
df=pd.DataFrame(te.fit(transactions).transform(transactions),
                columns=te.columns_)
for support in [.3,.4]:
    freq=apriori(df,min_support=support,use_colnames=True)
    rules=association_rules(freq,metric="confidence",min_threshold=.5)
    print("\nSupport:",support)
    print(freq)
    print(rules[["antecedents","consequents","support","confidence"]])
