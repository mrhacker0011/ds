Q1 [10 Marks] Clean employee data by handling missing values, duplicates and IQR outliers.
Solution — Q1
import pandas as pd
df=pd.DataFrame({
 "Emp_ID":[1,2,3,3,4,5],
 "Age":[22,25,None,25,32,55],
 "Dept":["IT","HR","IT","IT","Sales","IT"],
 "Years_Experience":[1,3,5,5,7,30],
 "Performance_Rating":[3,4,4,4,5,5]
})
df["Age"]=df["Age"].fillna(df["Age"].median())
df=df.drop_duplicates()
q1=df["Years_Experience"].quantile(.25)
q3=df["Years_Experience"].quantile(.75)
iqr=q3-q1
out=df[(df["Years_Experience"]<q1-1.5*iqr) |
       (df["Years_Experience"]>q3+1.5*iqr)]
print("Cleaned data:\n",df)
print("\nOutliers:\n",out)
Q2 [20 Marks] Use World Cities data for a geospatial map and population Treemap.
Solution — Q2 (Option A)
import pandas as pd
import folium
import matplotlib.pyplot as plt
import squarify
df=pd.read_csv("worldcities.csv").dropna(
    subset=["lat","lng","population"])
m=folium.Map(location=[20,0],zoom_start=2)
for _,r in df.head(100).iterrows():
    folium.CircleMarker(
        [r["lat"],r["lng"]],radius=5,
        popup=r["city"],fill=True).add_to(m)
m.save("world_cities_map.html")
top=df.nlargest(15,"population")
squarify.plot(sizes=top["population"],label=top["city"],alpha=.8)
plt.title("Population Treemap"); plt.axis("off"); plt.show()

Q2 [20 Marks] Use Crop Yield data for Simple Linear Regression from rainfall and evaluate MAE and R².
Solution — Q2 (Option B)
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error,r2_score
df=pd.read_csv("crop_yield_data.csv").dropna()
rain=[c for c in df.columns if "rain" in c.lower()][0]
target=[c for c in df.columns if "yield" in c.lower()][0]
X=df[[rain]]; y=df[target]
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,random_state=42)
model=LinearRegression().fit(Xtr,ytr)
pred=model.predict(Xte)
print("MAE:",mean_absolute_error(yte,pred))
print("R2:",r2_score(yte,pred))
plt.scatter(Xte,yte)
plt.plot(Xte,pred,color="red")
plt.xlabel("Rainfall"); plt.ylabel("Crop Yield")
plt.title("Rainfall vs Crop Yield")
plt.show()
