Q1 [10 Marks] Create product data and generate a Dendrogram.
Solution — Q1
import pandas as pd
import matplotlib.pyplot as plt
from scipy.cluster.hierarchy import dendrogram, linkage
df = pd.DataFrame({
    "Product":["P1","P2","P3","P4","P5"],
    "Price":[250,280,500,520,800],
    "Rating":[4.1,4.2,4.5,4.6,4.8],
    "Sales":[120,140,220,240,350]
})
Z = linkage(df[["Price","Rating","Sales"]], method="ward")
dendrogram(Z, labels=df["Product"].tolist())
plt.title("Product Similarity Dendrogram")
plt.xlabel("Products"); plt.ylabel("Distance")
plt.show()
Q2 [20 Marks] Apply Min-Max Scaling, Standardization and Normalization.
Solution — Q2 (Option A)
import pandas as pd
from sklearn.preprocessing import MinMaxScaler, StandardScaler, Normalizer
df = pd.DataFrame({
    "Temperature":[20,25,30,35,40],
    "Humidity":[45,50,60,65,70],
    "WindSpeed":[5,8,10,12,15],
    "Pressure":[1005,1010,1012,1015,1020],
    "Rainfall":[10,20,15,30,25]
})
print("Original:\n",df)
print("\nMin-Max:\n",pd.DataFrame(
    MinMaxScaler().fit_transform(df),columns=df.columns))
print("\nStandardized:\n",pd.DataFrame(
    StandardScaler().fit_transform(df),columns=df.columns))
print("\nNormalized:\n",pd.DataFrame(
    Normalizer().fit_transform(df),columns=df.columns))

Q2 [20 Marks] Apply KNN to Student Grading data and evaluate Accuracy, Precision, Recall, F1 and Confusion Matrix.
Solution — Q2 (Option B)
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score,precision_score,recall_score,f1_score,confusion_matrix
df=pd.read_csv("Students_Grading_Dataset.csv").dropna()
target=df.columns[-1]
X=pd.get_dummies(df.drop(columns=[target]),drop_first=True)
y=df[target].astype("category").cat.codes
X=StandardScaler().fit_transform(X)
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,
                                  random_state=42,stratify=y)
model=KNeighborsClassifier(n_neighbors=5).fit(Xtr,ytr)
pred=model.predict(Xte)
print("Accuracy:",accuracy_score(yte,pred))
print("Precision:",precision_score(yte,pred,average="weighted"))
print("Recall:",recall_score(yte,pred,average="weighted"))
print("F1:",f1_score(yte,pred,average="weighted"))
print("Confusion Matrix:\n",confusion_matrix(yte,pred))
