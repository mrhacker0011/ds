Q1 [10 Marks] Create a sales DataFrame from sales.csv and use a Box Plot to identify outliers.
Solution — Q1
import pandas as pd
import matplotlib.pyplot as plt
df=pd.read_csv("sales.csv")
sales=pd.to_numeric(df["Sales"],errors="coerce").dropna()
plt.boxplot(sales)
plt.title("Daily Sales Box Plot")
plt.ylabel("Sales")
plt.show()
Q2 [20 Marks] Apply KNN to the Wine dataset and evaluate Accuracy, Precision, Recall and F1.
Solution — Q2 (Option A)
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score,precision_score,recall_score,f1_score,confusion_matrix
df=pd.read_csv("Wine.csv").dropna()
X=df.iloc[:,:-1]; y=df.iloc[:,-1]
X=StandardScaler().fit_transform(X)
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,
                                  random_state=42,stratify=y)
model=KNeighborsClassifier(n_neighbors=5).fit(Xtr,ytr)
pred=model.predict(Xte)
print("Accuracy:",accuracy_score(yte,pred))
print("Precision:",precision_score(yte,pred))
print("Recall:",recall_score(yte,pred))
print("F1:",f1_score(yte,pred))
print("Confusion Matrix:\n",confusion_matrix(yte,pred))

Q2 [20 Marks] Apply Logistic Regression to Online Gaming data and evaluate classification metrics.
Solution — Q2 (Option B)
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score,precision_score,recall_score,f1_score,confusion_matrix
df=pd.read_csv("online_gaming_behavior_dataset.csv").dropna()
target=df.columns[-1]
X=pd.get_dummies(df.drop(columns=[target]),drop_first=True)
y=df[target].astype("category").cat.codes
X=StandardScaler().fit_transform(X)
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.2,
                                  random_state=42,stratify=y)
model=LogisticRegression(max_iter=1000).fit(Xtr,ytr)
pred=model.predict(Xte)
print("Accuracy:",accuracy_score(yte,pred))
print("Precision:",precision_score(yte,pred,average="weighted"))
print("Recall:",recall_score(yte,pred,average="weighted"))
print("F1:",f1_score(yte,pred,average="weighted"))
print("Confusion Matrix:\n",confusion_matrix(yte,pred))
